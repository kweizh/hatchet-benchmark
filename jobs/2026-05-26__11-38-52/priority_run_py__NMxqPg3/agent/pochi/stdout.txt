import os
import time
import threading
from hatchet_sdk import Hatchet, Context

run_id = os.environ.get("ZEALT_RUN_ID", "default")
task_name = f"prio-task-{run_id}"
log_file = f"/tmp/prio_log_{run_id}.txt"

# Ensure log file is truncated/created
with open(log_file, "w") as f:
    pass

hatchet = Hatchet()

@hatchet.task(name=task_name, default_priority=1)
def my_task(input: dict, ctx: Context):
    prio = input.get("priority", 1)
    epoch_ms = int(time.time() * 1000)
    with open(log_file, "a") as f:
        f.write(f"{prio} {epoch_ms}\n")
    time.sleep(2)
    return {"status": "ok"}

def start_worker():
    # Configure worker with slots=1
    worker = hatchet.worker("prio-worker", slots=1)
    worker.register_task(my_task)
    worker.start()

def main():
    # Start worker in background
    w_thread = threading.Thread(target=start_worker, daemon=True)
    w_thread.start()
    
    # Give the worker a moment to connect
    time.sleep(2)
    
    # Submit 5 task runs back-to-back with no waiting
    refs = []
    
    # First: 3 runs at priority 1 (low)
    for _ in range(3):
        ref = my_task.run(
            input={"priority": 1},
            priority=1,
            wait_for_result=False
        )
        refs.append(ref)
        
    # Then: 2 runs at priority 3 (high)
    for _ in range(2):
        ref = my_task.run(
            input={"priority": 3},
            priority=3,
            wait_for_result=False
        )
        refs.append(ref)
        
    # Wait for all 5 runs to complete
    for ref in refs:
        ref.result()
        
    print("All tasks completed.")

if __name__ == "__main__":
    main()
